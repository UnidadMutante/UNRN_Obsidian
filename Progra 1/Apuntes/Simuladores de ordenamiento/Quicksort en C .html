int partition(int arr[], int low, int high) {
int pivot = arr[high]; // Elegir pivote
int i = (low - 1); // Índice de menores
for (int j = low; j < high; j++) {
if (arr[j] <= pivot) { // Comparación
i++;
swap(&arr[i], &arr[j]); // Intercambio
}
}
swap(&arr[i + 1], &arr[high]); // Ubicar pivote
return (i + 1);
}
